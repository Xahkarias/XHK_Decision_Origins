@yearsBeforeDeadline = 1

# Storm Chasers
storm_chasers_quest = {
	length = 1
	icon = "GFX_edict_type_policy"
	resources = {
		category = edicts
	}
	potential = {
		AND = {
			is_ai = no
			years_passed < @yearsBeforeDeadline
			NOR = {
				has_country_flag = decision_storm_chasers
				has_origin = origin_storm_chasers
			}

			has_cosmic_storms_dlc = yes
			is_nomadic = no
		}
	}
	
	effect = {
		set_country_flag = decision_storm_chasers
		give_technology = { message = no tech = tech_storm_prediction_1 }
		owner_main_species = {
			change_species_characteristics = {				
				add_trait = trait_storm_touched
				add_traits_at_start_of_list = yes
			}
		}
		add_modifier = { modifier = storm_chaser_county_modifier }
		country_event = { id = cstorms.3101 days = 1 }
		country_event = { id = timeline_MMO.01 days = 4}
	}
	ai_weight = {
		weight = 0
	}
}

# Shroud-Forged 
shroud_forged_quest = {
	length = 1
	icon = "GFX_edict_type_policy"
	resources = {
		category = edicts
	}
	potential = {
		AND = {
			is_ai = no
			years_passed < 1
			NOT = {
				OR = {
					has_country_flag = decision_shroud_forged
					has_origin = origin_shroud_forged
				}
			}
			has_shroud_dlc = yes
		}
	}

	#allow = {
	#	custom_tooltip = {
	#		fail_text = "requires_machine_intelligence"
	#	}
	#}
	
	effect = {
		set_country_flag = decision_shroud_forged
		add_modifier = {
			modifier = shroud_forged_country_modifier
		}
		add_modifier = {
			modifier = robotic_species_trait_picks_modifier
		}
		add_modifier = {
			modifier = robotic_species_trait_points_modifier multiplier = 2
		}
		
		give_technology = { message = no tech = tech_mine_zro }
		add_resource = { sr_zro = 50 }
		capital_scope = {
			add_building = building_shroud_observatory_1
		}
		country_event = { id = shroud_forged.1 }
		country_event = { id = timeline_MMO.21 days = 4}
		
		owner_main_species = {
			drone_conversion_new_species_effect = yes
			set_species_flag = original_main_species # After conversion effect, so it's the only species with the flag
		}
		while = {
			create_pop_group = {
				size = 500
				species = event_target:secondary_main_species
				category = complex_drone
			}
		}
		
		if = {
			limit = {
				OR = {
					owner.capital_scope = {carrier_is_type = ship}
					is_machine_empire = no
				} 
			}
			country_event = { id = warning_event.01 }
		}
	}
	ai_weight = {
		weight = 0
	}
}
