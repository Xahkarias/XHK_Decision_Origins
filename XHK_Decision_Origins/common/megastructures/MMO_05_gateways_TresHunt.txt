# Gateway in Captain Ness treasure system
gateway_ness_restored = {
	entity = "reactivated_gateway_entity"
	construction_entity = "gateway_portal_entity"
	portrait = "GFX_megastructure_construction_background"
	place_entity_on_planet_plane = no
	show_galactic_map_icon = no
	show_in_outliner = no

	potential = {
		has_technology = tech_gateway_activation
	}

	upgrade_from = {
		gateway_ness_ruined
	}
	build_time = 720
	resources = {
		category = megastructures_gateways
		cost = {
			energy = 6000
		}
		cost = {
			trigger = {
				country_uses_bio_ships = no
			}
			alloys = 2500
		}
		cost = {
			trigger = {
				country_uses_bio_ships = yes
			}
			alloys = 2500
			mult = 0.5
		}
		cost = {
			trigger = {
				country_uses_bio_ships = yes
			}
			food = 2500
			mult = @halved_alloy_to_food_cost_ratio
		}
	}
	construction_blocks_and_blocked_by = self_type
	dismantle_potential = { always = no }

	bypass_type = gateway_ness
	on_build_complete = {
		activate_gateway = fromfrom
		fromfrom = { save_event_target_as = ness_gate }
		if = {
			limit = {
				has_grand_archive_dlc = yes
				exists = from
				from = {
					OR = {
						has_country_flag = decision_treasure_hunters
						has_origin = origin_treasure_hunters
					}
					has_special_project = ACTIVATE_GATEWAY_PROJECT
				}
				fromfrom = { has_megastructure_flag = gateway_to_ness_for@root.from }
			}
			from = {
				if = {
					limit = { has_special_project = ACTIVATE_GATEWAY_PROJECT }
					complete_special_project = {
						type = "ACTIVATE_GATEWAY_PROJECT"
						location = from
					}
				}
				spawn_ness_treasure_system = yes
				country_event = {
					id = grand_archive.8400
					scopes = { from = root.fromfrom }
				}
			}
		}
		if = {
			limit = {
				any_playable_country = {
					OR = {
						has_country_flag = decision_treasure_hunters
						has_origin = origin_treasure_hunters
					}
					is_nomadic = yes
					has_country_flag = activating_ness_gate
				}
			}
			random_playable_country = {
				limit = {
					OR = {
						has_country_flag = decision_treasure_hunters
						has_origin = origin_treasure_hunters
					}
					is_nomadic = yes
					has_country_flag = activating_ness_gate
				}
				spawn_ness_treasure_system = yes
				country_event = {
					id = grand_archive.8400
					scopes = { from = event_target:ness_gate}
				}

			}
		}
	}
}
