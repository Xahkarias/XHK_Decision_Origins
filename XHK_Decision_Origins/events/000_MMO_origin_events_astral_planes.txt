################################
# Astral Planes Origins Events #
################################

namespace = astral_planes

# Newly colonized planets gains Rift Affinity modifier if there's an astral rift
carrier_event = {
	id = astral_planes.2150
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		is_colony = yes
		owner = {
			OR = {
				has_country_flag = decision_riftworld
				has_origin = origin_riftworld
			}
		}
		solar_system = { has_astral_rift = yes }
		NOT = { has_modifier = riftworld_affinity }
	}

	immediate = {
		add_modifier = { modifier = riftworld_affinity }
	}
}

# Newly colonized planets gains Rift Affinity modifier if there's an astral rift
astral_rift_event = {
	id = astral_planes.2151
	hide_window = yes
	is_triggered_only = yes

	trigger = {
		solar_system = {
			exists = owner
			owner = {
				OR = {
					has_country_flag = decision_riftworld
					has_origin = origin_riftworld
				}
			}
		}
	}

	immediate = {
		solar_system = {
			every_system_planet_colony = {
				limit = {
					is_owned_by = prev.owner
					NOT = { has_modifier = riftworld_affinity }
				}
				add_modifier = { modifier = riftworld_affinity }
			}
		}
	}
}
